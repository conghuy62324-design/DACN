import {useEffect,useState} from 'react';
import {api} from '../../services/api';
import {safeWebLink,formatCostPrice,costCategoryLabels,costUnitLabels} from '../../utils/webSourceCards';
const empty={domain:'',publisher:'',sourceClass:'UNKNOWN',homepage:'',method:'ARTICLE',endpoint:'',articleUrls:[] as string[],permissionUrl:'',license:'UNVERIFIED',identityEvidence:'',robotsText:'',robotsReviewedAt:'',intervalHours:6,rateSeconds:10,retentionDays:30};
export default function TravelWebPage(){const[data,setData]=useState<any>(),[error,setError]=useState(''),[busy,setBusy]=useState(false),[draft,setDraft]=useState<any>(empty),[editing,setEditing]=useState<any>(),[reason,setReason]=useState(''),[confirmed,setConfirmed]=useState(false),[expires,setExpires]=useState(''),[imports,setImports]=useState<Record<string,string>>({}),[correction,setCorrection]=useState<any>(),[correctionField,setCorrectionField]=useState('party'),[correctionValue,setCorrectionValue]=useState(''),[evidence,setEvidence]=useState(''),[costEdit,setCostEdit]=useState<any>();
 const load=()=>api.get('/admin/travel-web').then((r:any)=>setData(r.data));useEffect(()=>{void load().catch(()=>setError('Chưa tải được quản trị web knowledge.'));},[]);
 const action=async(path:string,body:any,method='post')=>{if(busy)return;setBusy(true);setError('');try{await api.request({url:'/admin/travel-web'+path,method,data:body});await load();setConfirmed(false);}catch{setError('Chưa thực hiện được. Kiểm quyền, chế độ chỉ đọc, version và điều kiện nguồn; tải lại trước khi xác nhận.');}finally{setBusy(false);}};
 return <main className="space-y-5"><h1 className="text-2xl font-bold">Nguồn web & kiến thức du lịch</h1><p>Worker chỉ tạo candidate. Chỉ SUPER_ADMIN duyệt nguồn và kiến thức; chỉnh sửa nguồn yêu cầu duyệt lại. Không dùng chi phí bài viết làm báo giá dịch vụ.</p><p>Chế độ: {data?.status||'đang tải'}. Lịch discovery tối thiểu6h; refresh bài24h; worker mặc định tắt.</p>{error&&<p role="alert">{error}</p>}
 <section className="space-y-3 rounded border bg-white p-4"><h2 className="font-bold">{editing?'Sửa nguồn (cần duyệt lại)':'Thêm nguồn chưa duyệt'}</h2><div className="grid gap-3 md:grid-cols-2">{[['domain','Domain chính xác'],['publisher','Nhà xuất bản'],['homepage','Trang chủ HTTPS'],['permissionUrl','Liên kết quyền sử dụng/giấy phép'],['license','Giấy phép và nghĩa vụ ghi nguồn'],['identityEvidence','Bằng chứng danh tính nguồn'],['robotsReviewedAt','Robots đã kiểm lúc (ISO)']].map(([key,label])=><label key={key}>{label}<input className="block w-full rounded border p-2" value={draft[key]} onChange={e=>setDraft({...draft,[key]:e.target.value})}/></label>)}<label>Loại nguồn<select className="block border p-2" value={draft.sourceClass} onChange={e=>setDraft({...draft,sourceClass:e.target.value})}>{['OFFICIAL','EDITORIAL','COMMUNITY_EXPERIENCE','COMMERCIAL','UNKNOWN'].map(s=><option key={s}>{s}</option>)}</select></label><label>Phương thức<select value={draft.method} onChange={e=>setDraft({...draft,method:e.target.value})}>{['ARTICLE','RSS','API','SITEMAP'].map(s=><option key={s}>{s}</option>)}</select></label><label>RSS/API/Sitemap endpoint<input className="block w-full border p-2" value={draft.endpoint} onChange={e=>setDraft({...draft,endpoint:e.target.value})}/></label>{[['intervalHours','Chu kỳ discovery (giờ)'],['rateSeconds','Khoảng cách request (giây)'],['retentionDays','Retention candidate (ngày)']].map(([key,label])=><label key={key}>{label}<input type="number" value={draft[key]} onChange={e=>setDraft({...draft,[key]:Number(e.target.value)})}/></label>)}</div><label className="block">Tối đa3 URL bài được phép<textarea className="block w-full border p-2" value={draft.articleUrls.join('\n')} onChange={e=>setDraft({...draft,articleUrls:e.target.value.split('\n').filter(Boolean)})}/></label><label className="block">robots.txt đã kiểm<textarea className="block w-full border p-2" value={draft.robotsText} onChange={e=>setDraft({...draft,robotsText:e.target.value})}/></label><button disabled={busy||data?.status!=='SCOPED_WRITE'} onClick={()=>void action(editing?'/sources/'+editing.id:'/sources',{...draft,...(draft.endpoint?{}:{endpoint:undefined}),...(editing?{version:editing.version}:{})},editing?'patch':'post')} className="rounded border px-3 py-2">Lưu bản chưa duyệt</button><button onClick={()=>{setEditing(undefined);setDraft(empty);}} className="ml-3 underline">Nguồn mới</button></section>
 <section className="space-y-2"><h2 className="font-bold">Xác nhận quản trị</h2><input aria-label="Lý do xét duyệt web" className="w-full rounded border p-2" placeholder="Lý do và phạm vi đã kiểm" maxLength={240} value={reason} onChange={e=>setReason(e.target.value)}/><label className="flex gap-2"><input type="checkbox" checked={confirmed} onChange={e=>setConfirmed(e.target.checked)}/>Tôi đã kiểm quyền, nguồn gốc, facts, đơn vị chi phí và phạm vi sử dụng</label><label>Kiến thức hết hạn vào<input aria-label="Web knowledge expiry" type="datetime-local" value={expires} onChange={e=>setExpires(e.target.value)}/></label></section>
 <section><h2 className="font-bold">Danh sách nguồn</h2>{data?.sources.map((s:any)=><article key={s.id} className="my-3 rounded border bg-white p-3"><p className="font-semibold">{s.config.publisher} · {s.domain}</p><p>{s.review_status} · {s.permission} · {s.enabled?'đang bật':'đã tắt'} · version{s.version}</p><p>Fetch tốt: {s.last_success||'chưa có'} · Lỗi: {s.last_error||'không'}</p><button className="underline" onClick={()=>{setDraft({...s.config,endpoint:s.config.endpoint||''});setEditing(s);}}>Sửa nguồn</button>{[['APPROVE','Duyệt/bật'],['DISABLE','Tắt'],['REVOKE','Thu hồi']].map(([d,label])=><button key={d} disabled={busy||!confirmed||!reason.trim()||data.status!=='SCOPED_WRITE'} className="ml-3 rounded border p-2" onClick={()=>void action('/sources/'+s.id+'/moderation',{decision:d,version:s.version,reason,confirmed})}>{label}</button>)}<button disabled={busy||!s.enabled||data.status!=='SCOPED_WRITE'} className="ml-3 underline" onClick={()=>void action('/sources/'+s.id+'/refresh',{})}>Yêu cầu cập nhật có giới hạn</button><label className="mt-2 block">Nhập URL bài đã được phép<input aria-label={'URL bài '+s.domain} className="block w-full border p-2" value={imports[s.id]||''} onChange={e=>setImports({...imports,[s.id]:e.target.value})}/></label><button disabled={busy||!confirmed||!imports[s.id]||!s.enabled||data.status!=='SCOPED_WRITE'} className="rounded border p-2" onClick={()=>void action('/sources/'+s.id+'/import',{url:imports[s.id],version:s.version,confirmed})}>Nhập bài chờ xử lý</button><details><summary>Lịch sử version</summary>{s.history.map((h:any,i:number)=><p key={i}>Version{h.version} · {h.decision||'sửa cấu hình'} · {h.reason||h.at}</p>)}</details></article>)}</section>
 <section><h2 className="font-bold">Bài viết/candidate mới</h2>{data?.candidates.map((c:any)=><article key={c.id} className="my-3 space-y-2 rounded border bg-white p-3"><p className="font-semibold">{c.extraction.title}</p><p>{c.status} · {c.quality} · version{c.version}</p>{safeWebLink(c.canonical_url)&&<a className="underline" href={c.canonical_url} target="_blank" rel="noopener noreferrer">Mở bài gốc</a>}<p>Ngày đăng {c.extraction.publishedAt||'chưa rõ'} · Chuyến đi {c.extraction.travelPeriod||'chưa rõ'} · Truy xuất {c.extraction.retrievedAt}</p>{c.extraction.facts.map((f:any,i:number)=><p key={i}>{f.summary}</p>)}{c.extraction.costs.map((o:any,i:number)=><p key={i}>Chi phí nguồn: {formatCostPrice(o)} · {costCategoryLabels[o.costCategory||o.category]}. Bao gồm {o.included.join(', ')||'chưa xác định'}; chưa gồm {o.excluded.join(', ')||'UNKNOWN'}</p>)}<button className="rounded border p-2" disabled={busy||data.status!=='SCOPED_WRITE'} onClick={()=>{setCorrection(c);setEvidence(c.canonical_url);setCorrectionValue('');setCostEdit(undefined);}}>Sửa dữ liệu theo bài gốc</button>{c.extraction.costs.map((o:any,i:number)=><button key={i} className="mr-2 rounded border p-2" disabled={busy||data.status!=='SCOPED_WRITE'} onClick={()=>{setCorrection(c);setEvidence(c.canonical_url);setCostEdit({...o,amountMin:o.amountMin??o.amount,amountMax:o.amountMax??o.amount,priceBound:o.priceBound||'EXACT'});}}>Sửa khoản nguồn {i+1}</button>)}<details><summary>Lịch sử sửa dữ liệu</summary>{c.extraction.reviewRevisions?.map((r:any,i:number)=><p key={i}>Bản {r.version} · {r.reason} · {r.evidenceReference}</p>)}</details>{[['APPROVE','Duyệt kiến thức'],['REJECT','Từ chối'],['REVOKE','Thu hồi']].map(([d,label])=><button key={d} disabled={busy||!confirmed||!reason.trim()||data.status!=='SCOPED_WRITE'||d==='APPROVE'&&!expires} className="mr-3 rounded border p-2" onClick={()=>void action('/candidates/'+c.id+'/review',{decision:d,version:c.version,reason,confirmed,...(d==='APPROVE'?{expiresAt:new Date(expires).toISOString()}:{})})}>{label}</button>)}</article>)}</section>{correction&&<section className="space-y-2 rounded border p-4"><h2 className="font-bold">Sửa candidate — phải duyệt lại</h2>{!costEdit&&<label>Field nguồn<select value={correctionField} onChange={e=>setCorrectionField(e.target.value)}>{['origin','days','nights','party','rooms','travelPeriod','transportation','accommodationType','travelStyle'].map(f=><option key={f}>{f}</option>)}</select></label>}{!costEdit&&<label className="block">Giá trị nguồn (để trống nếu chưa xác định)<input aria-label="Giá trị sửa nguồn" className="block w-full border p-2" value={correctionValue} onChange={e=>setCorrectionValue(e.target.value)}/></label>}{costEdit&&<div className="grid gap-2 md:grid-cols-2"><label>Phạm vi số tiền<select value={costEdit.priceBound} onChange={e=>setCostEdit({...costEdit,priceBound:e.target.value})}>{['EXACT','RANGE','AT_MOST','AT_LEAST'].map(x=><option key={x}>{x}</option>)}</select></label>{['amountMin','amountMax'].map(k=><label key={k}>{k==='amountMin'?'Số tiền/cận dưới':'Cận trên'}<input type="number" min="0" max="1000000000" className="block w-full border p-2" value={costEdit[k]??''} onChange={e=>setCostEdit({...costEdit,[k]:e.target.value===''?null:Number(e.target.value)})}/></label>)}<label>Tiền tệ<select value={costEdit.currency} onChange={e=>setCostEdit({...costEdit,currency:e.target.value})}>{['VND','USD','EUR','GBP','UNKNOWN'].map(x=><option key={x}>{x}</option>)}</select></label><label>Nhóm khoản<select value={costEdit.costCategory} onChange={e=>setCostEdit({...costEdit,costCategory:e.target.value})}>{Object.entries(costCategoryLabels).filter(([k])=>!['TOTAL','SIGHTSEEING','RESERVE'].includes(k)).map(([k,v])=><option value={k} key={k}>{v}</option>)}</select></label><label>Đơn vị<select value={costEdit.unit} onChange={e=>setCostEdit({...costEdit,unit:e.target.value})}>{Object.entries(costUnitLabels).map(([k,v])=><option value={k} key={k}>{v}</option>)}</select></label></div>}<label className="block">Evidence trong đúng bài gốc<input aria-label="Evidence sửa nguồn" className="block w-full border p-2" value={evidence} onChange={e=>setEvidence(e.target.value)}/></label><button disabled={busy||!confirmed||!reason.trim()||!evidence||data?.status!=='SCOPED_WRITE'} className="rounded border p-2" onClick={()=>void action('/candidates/'+correction.id+'/extraction',{version:correction.version,confirmed,reason,evidenceReference:evidence,changes:costEdit?{costs:[{observationId:costEdit.observationId,priceBound:costEdit.priceBound,amount:costEdit.priceBound==='EXACT'?costEdit.amountMin:null,amountMin:costEdit.priceBound==='AT_MOST'?null:costEdit.amountMin,amountMax:costEdit.priceBound==='AT_LEAST'?null:costEdit.priceBound==='EXACT'?costEdit.amountMin:costEdit.amountMax,currency:costEdit.currency,unit:costEdit.unit,costCategory:costEdit.costCategory}]}:{[correctionField]:!correctionValue.trim()?null:['days','nights','party','rooms'].includes(correctionField)?Number(correctionValue):correctionValue}},'patch').then(()=>setCorrection(undefined))}>Lưu bản sửa chờ duyệt</button></section>}<section><h2 className="font-bold">Job cập nhật</h2>{data?.jobs.map((j:any)=><p key={j.id}>Lần tới {j.next_due} · {j.kind} · {j.last_error||'không lỗi'} · attempts{j.attempts}</p>)}</section></main>;
}
