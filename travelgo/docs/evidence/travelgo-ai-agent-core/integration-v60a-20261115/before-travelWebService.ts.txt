export const webIntelligenceEnabled=()=>process.env.TRAVEL_WEB_ENABLED==='true';
export const webWritesAllowed=()=>webIntelligenceEnabled()&&process.env.TRAVEL_WEB_WRITES_ENABLED==='true'&&process.env.ADVISORY_STAGING_READ_ONLY!=='true';
export const requireWebAdmin=(actor:{id:string;roles:string[]})=>{if(!actor?.roles?.includes('SUPER_ADMIN'))throw new HttpError(403,'Chỉ SUPER_ADMIN được quản lý nguồn và duyệt kiến thức.','FORBIDDEN');};
function writable(){if(!webWritesAllowed())throw new HttpError(403,'Web knowledge đang chỉ đọc.','WEB_READ_ONLY');}

const sourceRow=(r:any):SourceRecord=>({...r.config,id:r.id,version:r.version,permission:r.permission,reviewStatus:r.review_status,enabled:r.enabled});

 async candidateHistory(actor:{id:string;roles:string[]},id:string){
  requireWebAdmin(actor);if(!webIntelligenceEnabled())return{revisions:[],status:'WEB_DISABLED'};
  if(!/^[0-9a-f-]{36}$/i.test(id))throw new HttpError(404,'Không tìm thấy candidate.','WEB_CANDIDATE_NOT_FOUND');
  const candidate=(await select('SELECT id,knowledge_id,status FROM travel_web_candidates WHERE id=$1',[id]))[0];if(!candidate)throw new HttpError(404,'Không tìm thấy candidate.','WEB_CANDIDATE_NOT_FOUND');
  const rows=await select("SELECT id,version,status,reviewed_at,created_at,provenance FROM verified_travel_knowledge WHERE provenance->>'webCandidateId'=$1 ORDER BY created_at DESC,id LIMIT 10",[id]);
  return{status:'READ_ONLY_HISTORY',revisions:rows.map(k=>({id:k.id,version:k.version,status:k.status,reviewedAt:k.reviewed_at,createdAt:k.created_at,sourceVersion:k.provenance.webSourceVersion,candidateVersion:k.provenance.webCandidateVersion,currentSnapshot:candidate.status==='APPROVED'&&candidate.knowledge_id===k.id,snapshot:{party:k.provenance.external?.party??null,days:k.provenance.external?.days??null,nights:k.provenance.external?.nights??null,costs:k.provenance.external?.costs||[]}}))};
 }
 async importArticle(actor:{id:string;roles:string[]},sourceId:string,raw:any){requireWebAdmin(actor);writable();if(!raw||Object.keys(raw).some(k=>!['url','version','confirmed'].includes(k))||typeof raw.url!=='string'||!Number.isSafeInteger(raw.version)||raw.confirmed!==true)throw new HttpError(400,'Cần URL,version và xác nhận nhập bài.','WEB_IMPORT_INVALID');return sequelize.transaction(async transaction=>{const row=(await select('SELECT * FROM travel_web_sources WHERE id=$1 FOR UPDATE',[sourceId],transaction))[0];if(!row||row.version!==raw.version)throw new HttpError(409,'Nguồn vừa thay đổi.','WEB_REVISION_CONFLICT');const source=sourceRow(row),url=allowedSourceUrl(source,raw.url);return{jobs:await select("INSERT INTO travel_web_jobs(source_id,url,kind) VALUES($1,$2,'ARTICLE') ON CONFLICT(source_id,url) DO NOTHING RETURNING id",[sourceId,url],transaction),message:'URL đã đi qua permission/robots policy; chỉ enqueue, chưa đọc hoặc duyệt bài.'};});}
 async correctCandidate(actor:{id:string;roles:string[]},id:string,raw:any){requireWebAdmin(actor);writable();if(!raw||Object.keys(raw).some(k=>!['version','confirmed','reason','evidenceReference','changes'].includes(k))||raw.confirmed!==true||!Number.isSafeInteger(raw.version)||!safeKnowledgeText(raw.reason,240)||typeof raw.evidenceReference!=='string'||raw.evidenceReference.length>600||!raw.changes||Array.isArray(raw.changes)||Object.keys(raw.changes).some(k=>!['origin','days','nights','party','rooms','travelPeriod','transportation','accommodationType','travelStyle','costs'].includes(k)))throw new HttpError(400,'Bản sửa cần field có kiểu,reason và evidence.','WEB_CORRECTION_INVALID');
  return sequelize.transaction(async transaction=>{const ref=(await select('SELECT source_id FROM travel_web_candidates WHERE id=$1',[id],transaction))[0];if(!ref)throw new HttpError(404,'Không tìm thấy candidate.','WEB_CANDIDATE_NOT_FOUND');const sr=(await select('SELECT * FROM travel_web_sources WHERE id=$1 FOR UPDATE',[ref.source_id],transaction))[0],row=(await select('SELECT * FROM travel_web_candidates WHERE id=$1 FOR UPDATE',[id],transaction))[0];if(!sr||!row||row.version!==raw.version||row.source_version!==sr.version||!sourceUsable(sourceRow(sr)))throw new HttpError(409,'Nguồn hoặc candidate vừa thay đổi.','WEB_REVISION_CONFLICT');if(!raw.evidenceReference.startsWith(row.canonical_url+'#')&&raw.evidenceReference!==row.canonical_url)throw new HttpError(400,'Evidence phải trỏ đúng bài gốc.','WEB_CORRECTION_EVIDENCE');const x=structuredClone(row.extraction) as WebExtraction,change=raw.changes;
   for(const key of ['days','nights','party','rooms'])if(key in change){if(change[key]!==null&&(!Number.isInteger(change[key])||change[key]<(key==='nights'?0:1)||change[key]>100))throw new HttpError(400,'Số lượng không hợp lệ.','WEB_CORRECTION_INVALID');(x as any)[key]=change[key];}
   for(const key of ['origin','travelPeriod','transportation','accommodationType','travelStyle'])if(key in change){if(change[key]!==null&&!safeKnowledgeText(change[key],100))throw new HttpError(400,'Field nguồn không hợp lệ.','WEB_CORRECTION_INVALID');if(key==='origin'&&change[key]&&!resolveLocation(change[key]))throw new HttpError(400,'Dùng shared geography.','WEB_CORRECTION_INVALID');(x as any)[key]=change[key];}
   if(change.costs!==undefined){if(!Array.isArray(change.costs)||!change.costs.length||change.costs.length>12)throw new HttpError(400,'Bản sửa chi phí không hợp lệ.','WEB_CORRECTION_INVALID');for(const edit of change.costs){if(!edit||Object.keys(edit).some(k=>!['observationId','amount','amountMin','amountMax','currency','unit','costCategory','included','excluded','unknownItems','priceBound'].includes(k)))throw new HttpError(400,'Không được thêm field ngoài contract.','WEB_CORRECTION_INVALID');const at=x.costs.findIndex(o=>o.observationId===edit.observationId);if(at<0)throw new HttpError(400,'Observation không có trong evidence.','WEB_CORRECTION_INVALID');const next={...x.costs[at],...edit} as CostObservation;if(!COST_UNITS.includes(next.unit)||!COST_CATEGORIES.includes(next.costCategory)||!['VND','USD','EUR','GBP','UNKNOWN'].includes(next.currency)||next.amount!==null&&(!Number.isFinite(next.amount)||next.amount<0||next.amount>1e9)||[next.amountMin,next.amountMax].some(v=>v!==null&&v!==undefined&&(!Number.isFinite(v)||v<0||v>1e9))||next.amountMin!==null&&next.amountMax!==null&&next.amountMin>next.amountMax||[next.included,next.excluded,next.unknownItems||[]].some(v=>!Array.isArray(v)||v.length>10||v.some(t=>!safeKnowledgeText(t,90))))throw new HttpError(400,'Kiểu,đơn vị hoặc khoảng tiền chưa hợp lệ.','WEB_CORRECTION_INVALID');if(!typedCostObservation(next))throw new HttpError(400,'Observation không đúng hợp đồng kiểu/khoảng.','WEB_CORRECTION_INVALID');x.costs[at]={...next,category:next.costCategory==='TOTAL_TRIP'?'TOTAL':next.costCategory==='ATTRACTION'?'SIGHTSEEING':next.costCategory};}}
   if(x.days&&x.nights!==null&&x.nights>=x.days)throw new HttpError(400,'Ngày/đêm mâu thuẫn.','WEB_CORRECTION_INVALID');x.reviewRevisions=[...(x.reviewRevisions||[]),{version:row.version,changes:change,reason:raw.reason,evidenceReference:raw.evidenceReference,reviewedAt:new Date().toISOString()}].slice(-10);x.costs=x.costs.map(o=>({...o,candidateVersion:row.version+1}));x.manualCorrection={method:'HUMAN_SOURCE_CORRECTION',actorId:actor.id,evidenceReference:raw.evidenceReference};if(Buffer.byteLength(JSON.stringify(x))>100000)throw new HttpError(400,'Revision vượt giới hạn.','WEB_CORRECTION_INVALID');const updated=(await select("UPDATE travel_web_candidates SET extraction=$2::jsonb,status='DRAFT',quality='UNREVIEWED',version=version+1,updated_at=now() WHERE id=$1 RETURNING *",[id,JSON.stringify(x)],transaction))[0];await select("INSERT INTO audit_logs(actor_user_id,action,entity_type,entity_id,previous_data,new_data,reason) VALUES($1,'WEB_CANDIDATE_CORRECTED','TravelWebCandidate',$2,$3::jsonb,$4::jsonb,$5) RETURNING id",[actor.id,id,JSON.stringify({version:row.version,extraction:row.extraction}),JSON.stringify({version:updated.version,changes:change,evidenceReference:raw.evidenceReference}),raw.reason],transaction);return updated;});
 }
 async sources(enabledOnly=false):Promise<SourceRecord[]>{if(!webIntelligenceEnabled())return[];return(await select(`SELECT * FROM travel_web_sources ${enabledOnly?"WHERE enabled AND permission='ALLOWED' AND review_status='APPROVED'":''} ORDER BY domain LIMIT 50`)).map(sourceRow);}
 async adminOverview(actor:{id:string;roles:string[]}){requireWebAdmin(actor);if(!webIntelligenceEnabled())return{sources:[],jobs:[],candidates:[],status:'WEB_DISABLED'};return{sources:await select('SELECT * FROM travel_web_sources ORDER BY domain LIMIT 50'),jobs:await select('SELECT * FROM travel_web_jobs ORDER BY updated_at DESC LIMIT 50'),candidates:await select('SELECT * FROM travel_web_candidates ORDER BY updated_at DESC LIMIT 50'),status:webWritesAllowed()?'SCOPED_WRITE':'READ_ONLY'};}

  return sequelize.transaction(async transaction=>{const old=id?(await select('SELECT * FROM travel_web_sources WHERE id=$1 FOR UPDATE',[id],transaction))[0]:null;if(id&&(!old||old.version!==version))throw new HttpError(409,'Nguồn vừa thay đổi.','WEB_REVISION_CONFLICT');
   const history=old?[...(old.history||[]),{version:old.version,config:old.config,permission:old.permission,reviewStatus:old.review_status,at:old.updated_at}].slice(-10):[];

 async moderateSource(actor:{id:string;roles:string[]},id:string,raw:any){requireWebAdmin(actor);writable();if(!raw||Object.keys(raw).some(k=>!['version','decision','reason','confirmed'].includes(k))||!Number.isSafeInteger(raw.version)||!['APPROVE','DISABLE','REVOKE'].includes(raw.decision)||raw.confirmed!==true||!safeKnowledgeText(raw.reason,240))throw new HttpError(400,'Cần xác nhận và lý do duyệt nguồn.','WEB_MODERATION_INVALID');
  return sequelize.transaction(async transaction=>{const old=(await select('SELECT * FROM travel_web_sources WHERE id=$1 FOR UPDATE',[id],transaction))[0];if(!old||old.version!==raw.version)throw new HttpError(409,'Nguồn vừa thay đổi.','WEB_REVISION_CONFLICT');const s=sourceDraft(old.config);
   if(raw.decision==='APPROVE'&&(s.sourceClass==='UNKNOWN'||s.license==='UNVERIFIED'||Date.now()-Date.parse(s.robotsReviewedAt)>7*86400000||![s.endpoint,...s.articleUrls].filter(Boolean).every(url=>{try{allowedSourceUrl({...s,id,version:old.version,enabled:true,permission:'ALLOWED',reviewStatus:'APPROVED'},url!);return true;}catch{return false;}})))throw new HttpError(400,'Quyền/robots của nguồn chưa đủ.','WEB_SOURCE_PERMISSION_REQUIRED');
   const row=(await select('UPDATE travel_web_sources SET enabled=$2,permission=$3,review_status=$4,reviewed_by=$5,version=version+1,history=$6::jsonb,updated_at=now() WHERE id=$1 RETURNING *',[id,raw.decision==='APPROVE',raw.decision==='APPROVE'?'ALLOWED':raw.decision==='REVOKE'?'DENIED':old.permission,raw.decision==='APPROVE'?'APPROVED':raw.decision==='REVOKE'?'REVOKED':old.review_status,actor.id,JSON.stringify([...(old.history||[]),{version:old.version,decision:raw.decision,reason:raw.reason,actorId:actor.id,at:new Date().toISOString()}].slice(-10))],transaction))[0];return row;});
 }
 async enqueue(actor:{id:string;roles:string[]},sourceId:string){requireWebAdmin(actor);writable();const source=(await this.sources()).find(s=>s.id===sourceId);if(!source||!sourceUsable(source))throw new HttpError(400,'Nguồn chưa được duyệt.','WEB_SOURCE_NOT_APPROVED');return this.enqueueSource(source);}
 async enqueueSource(source:SourceRecord,transaction?:Transaction){writable();if(!sourceUsable(source))return[];const urls=source.method==='ARTICLE'?source.articleUrls:[source.endpoint!];const out=[];for(const url of urls){allowedSourceUrl(source,url);out.push(...await select('INSERT INTO travel_web_jobs(source_id,url,kind) VALUES($1,$2,$3) ON CONFLICT(source_id,url) DO NOTHING RETURNING id',[source.id,url,source.method==='ARTICLE'?'ARTICLE':'DISCOVER'],transaction));}return out;}
 async claim(now=new Date(),leaseSeconds=30){writable();return sequelize.transaction(async transaction=>{
  const gate=await select("SELECT pg_try_advisory_xact_lock(hashtext('travelgo_web_global_claim')) AS ok",[],transaction);if(!gate[0].ok||(await select('SELECT count(*)::int AS n FROM travel_web_jobs WHERE lease_until>$1',[now],transaction))[0].n>=2)return null;
  const rows=await select("SELECT j.*,s.config,s.version AS source_version,s.permission,s.review_status,s.enabled,s.last_success,s.last_attempt FROM travel_web_jobs j JOIN travel_web_sources s ON s.id=j.source_id WHERE j.next_due<=$1 AND (j.lease_until IS NULL OR j.lease_until<=$1) AND s.enabled AND s.permission='ALLOWED' AND s.review_status='APPROVED' AND (s.last_attempt IS NULL OR s.last_attempt<=$1-((s.config->>'rateSeconds')::integer*interval '1 second')) ORDER BY j.next_due,j.id FOR UPDATE OF j,s SKIP LOCKED LIMIT 1",[now],transaction),job=rows[0];if(!job)return null;
  const source=sourceRow({...job,id:job.source_id,version:job.source_version});if(!sourceUsable(source,+now)||job.last_success&&+now-Date.parse(job.last_success)<source.rateSeconds*1000)return null;
  const token=randomUUID();await select('UPDATE travel_web_sources SET last_attempt=$2 WHERE id=$1 RETURNING id',[job.source_id,now],transaction);await select('UPDATE travel_web_jobs SET lease_token=$2,lease_until=$3,updated_at=now() WHERE id=$1 RETURNING id',[job.id,token,new Date(+now+Math.min(60,leaseSeconds)*1000)],transaction);return{...job,lease_token:token,source};});}
 async storeExtraction(job:any,x:WebExtraction){writable();return sequelize.transaction(async transaction=>{
  const source=(await select('SELECT * FROM travel_web_sources WHERE id=$1 FOR UPDATE',[job.source_id],transaction))[0],lease=(await select('SELECT * FROM travel_web_jobs WHERE id=$1 FOR UPDATE',[job.id],transaction))[0];if(!source||!lease||lease.lease_token!==job.lease_token||Date.parse(lease.lease_until)<=Date.now()||source.version!==job.source.version||!sourceUsable(sourceRow(source)))throw new HttpError(409,'Lease hoặc nguồn đã thay đổi.','WEB_SOURCE_CHANGED');
  allowedSourceUrl(sourceRow(source),x.canonicalUrl);const prior=await select('SELECT * FROM travel_web_candidates ORDER BY created_at DESC LIMIT 100',[],transaction);
  const duplicate=prior.find(r=>r.source_version===source.version&&duplicateArticle(r.extraction,x));if(duplicate)return{duplicate:true,id:duplicate.id};
  const quality=prior.some(r=>!duplicateArticle(r.extraction,x)&&conflictingCosts(r.extraction,x))?'CONFLICTING':x.quality;
  await select("UPDATE travel_web_candidates SET status='STALE',quality='STALE',version=version+1,updated_at=now() WHERE source_id=$1 AND canonical_url=$2 AND content_hash<>$3 AND status NOT IN ('REJECTED','STALE') RETURNING id",[source.id,x.canonicalUrl,x.contentHash],transaction);
  const candidateId=randomUUID(),stored={...x,costs:x.costs.map(o=>({...o,sourceId:source.id,candidateId,candidateVersion:1}))};const candidate=(await select('INSERT INTO travel_web_candidates(source_id,source_version,canonical_url,content_hash,extraction,quality,id) VALUES($1,$2,$3,$4,$5::jsonb,$6,$7) ON CONFLICT DO NOTHING RETURNING *',[source.id,source.version,x.canonicalUrl,x.contentHash,JSON.stringify(stored),quality,candidateId],transaction))[0];return{duplicate:!candidate,id:candidate?.id};});}

 async prune(){writable();return select("DELETE FROM travel_web_candidates WHERE id IN (SELECT c.id FROM travel_web_candidates c JOIN travel_web_sources s ON s.id=c.source_id WHERE c.knowledge_id IS NULL AND c.status IN ('DRAFT','REJECTED','STALE') AND c.created_at<now()-((s.config->>'retentionDays')::integer*interval '1 day') ORDER BY c.created_at LIMIT 20) RETURNING id");}
 async finish(job:any,result:{etag?:string;modified?:string;error?:string;hasCosts?:boolean},now=new Date()){
  writable();const attempts=result.error?Math.min(12,job.attempts+1):0,seconds=result.error?Math.min(86400,Math.max(job.source.rateSeconds,60*2**Math.min(10,attempts))):Math.max(job.source.rateSeconds,(job.kind==='DISCOVER'?job.source.intervalHours:result.hasCosts?168:24)*3600);
  return sequelize.transaction(async transaction=>{const rows=await select('UPDATE travel_web_jobs SET lease_until=NULL,lease_token=NULL,attempts=$3,last_error=$4,etag=coalesce($5,etag),last_modified=coalesce($6,last_modified),next_due=$7,updated_at=now() WHERE id=$1 AND lease_token=$2 RETURNING id',[job.id,job.lease_token,attempts,result.error||null,result.etag||null,result.modified||null,new Date(+now+seconds*1000)],transaction);if(rows.length)await select('UPDATE travel_web_sources SET last_success=CASE WHEN $2::text IS NULL THEN $3 ELSE last_success END,last_error=$2 WHERE id=$1 RETURNING id',[job.source_id,result.error||null,now],transaction);});
 }
 async reviewCandidate(actor:{id:string;roles:string[]},id:string,raw:any){requireWebAdmin(actor);writable();if(!raw||Object.keys(raw).some(k=>!['version','decision','reason','confirmed','expiresAt'].includes(k))||raw.confirmed!==true||!Number.isSafeInteger(raw.version)||!['APPROVE','REJECT','REVOKE'].includes(raw.decision)||!safeKnowledgeText(raw.reason,240)||raw.decision==='APPROVE'&&(!Number.isFinite(Date.parse(raw.expiresAt))||Date.parse(raw.expiresAt)<=Date.now()||Date.parse(raw.expiresAt)>Date.now()+90*86400000))throw new HttpError(400,'Cần xác nhận bản trích xuất và thời hạn.','WEB_REVIEW_INVALID');
  return sequelize.transaction(async transaction=>{
   const ref=(await select('SELECT source_id FROM travel_web_candidates WHERE id=$1',[id],transaction))[0];if(!ref)throw new HttpError(404,'Không tìm thấy candidate.','WEB_CANDIDATE_NOT_FOUND');await select('SELECT id FROM travel_web_sources WHERE id=$1 FOR UPDATE',[ref.source_id],transaction);const row=(await select('SELECT * FROM travel_web_candidates WHERE id=$1 FOR UPDATE',[id],transaction))[0];if(!row||row.version!==raw.version)throw new HttpError(409,'Candidate vừa thay đổi.','WEB_REVISION_CONFLICT');const source=(await select('SELECT * FROM travel_web_sources WHERE id=$1 FOR UPDATE',[row.source_id],transaction))[0];
   if(raw.decision!=='APPROVE'){if(row.knowledge_id){const k=(await select('SELECT version FROM verified_travel_knowledge WHERE id=$1',[row.knowledge_id],transaction))[0];await TravelKnowledgeService.write(actor,{status:'REVOKED',version:k.version,reason:raw.reason},row.knowledge_id,transaction);}return(await select('UPDATE travel_web_candidates SET status=$2,reviewed_by=$3,version=version+1,updated_at=now() WHERE id=$1 RETURNING *',[id,raw.decision==='REJECT'?'REJECTED':'STALE',actor.id],transaction))[0];}
   if(!source||row.source_version!==source.version||!sourceUsable(sourceRow(source))||row.status!=='DRAFT'||['STALE','CONFLICTING'].includes(row.quality))throw new HttpError(409,'Nguồn hoặc evidence chưa đủ điều kiện.','WEB_CANDIDATE_NOT_ELIGIBLE');
   const x=row.extraction as WebExtraction;allowedSourceUrl(sourceRow(source),x.canonicalUrl);
   const draft={locationId:x.locationId,topic:x.costs.length?'cost':'overview',title:x.title,content:candidateSummary(x),sourceType:'EXTERNAL_REVIEWED',sourceUrl:x.canonicalUrl,provenance:{method:'WEB_SOURCE_REVIEW',retrievedAt:x.retrievedAt,evidenceHash:x.contentHash,webSourceId:source.id,webCandidateId:id,webCandidateVersion:row.version,webSourceVersion:source.version,external:x,license:source.config.license,permissionUrl:source.config.permissionUrl,...(source.config.independenceGroup?{independenceGroup:source.config.independenceGroup,independenceReviewed:true}:{})},expiresAt:raw.expiresAt};
   const k=await TravelKnowledgeService.write(actor,draft,undefined,transaction);const approved=await TravelKnowledgeService.write(actor,{status:'APPROVED',version:k.version,publicUse:true,reason:raw.reason},k.id,transaction);
   return(await select("UPDATE travel_web_candidates SET status='APPROVED',knowledge_id=$2,reviewed_by=$3,version=version+1,updated_at=now() WHERE id=$1 RETURNING *",[id,approved.id,actor.id],transaction))[0];});
 }
