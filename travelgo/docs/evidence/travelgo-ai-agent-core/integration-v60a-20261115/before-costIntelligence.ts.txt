 const out:CostEvidence[]=[];for(const k of items.slice(0,100)){if(k.status!=='APPROVED'||!k.publicUse||!k.reviewedAt||!Number.isFinite(Date.parse(k.reviewedAt))||Date.parse(k.reviewedAt)>now||k.expiresAt&&(!Number.isFinite(Date.parse(k.expiresAt))||Date.parse(k.expiresAt)<=now)||k.validFrom&&(!Number.isFinite(Date.parse(k.validFrom))||Date.parse(k.validFrom)>now)||k.sourceType!=='EXTERNAL_REVIEWED'||k.provenance?.method!=='WEB_SOURCE_REVIEW')continue;const x=k.provenance.external as CostTrip;if(!x||!Array.isArray(x.costs))continue;
  for(const raw of x.costs.slice(0,12)){const o=typedCostObservation(raw);if(!o)continue;if(o.validityKind==='CURRENT_TARIFF'&&(!o.validUntil||!o.validFrom||!Number.isFinite(Date.parse(o.validUntil))||!Number.isFinite(Date.parse(o.validFrom))||Date.parse(o.validFrom)>now||Date.parse(o.validUntil)<=now))continue;const normalized=normalizeCost(o,x);out.push({knowledgeId:k.id,knowledgeVersion:k.version,sourceUrl:k.sourceUrl!,title:k.title,sourceKey:typeof k.provenance.independenceGroup==='string'&&k.provenance.independenceReviewed===true?k.provenance.independenceGroup:null,tripKey:x.originatingTripId||x.originatingUrl||x.canonicalUrl,trip:x,observation:o,normalized,sourceClass:String((x as any).sourceClass||'UNKNOWN'),license:String(k.provenance.license||''),synthetic:x.synthetic===true});}
 }return out;
}
export type Comparability={status:'COMPARABLE'|'PARTIALLY_COMPARABLE'|'NOT_COMPARABLE'|'INSUFFICIENT_DATA';categoryComparable:boolean;differences:string[];missing:string[]};
export function compareCosts(a:CostEvidence,b:CostEvidence,crossDestination=false):Comparability{
 const missing:string[]=[],differences:string[]=[],cat=a.observation.costCategory||({TOTAL:'TOTAL_TRIP',SIGHTSEEING:'ATTRACTION'}[a.observation.category]||a.observation.category),bcat=b.observation.costCategory||({TOTAL:'TOTAL_TRIP',SIGHTSEEING:'ATTRACTION'}[b.observation.category]||b.observation.category);
 if(cat!==bcat||a.normalized.currency!==b.normalized.currency||a.normalized.unit!==b.normalized.unit)return{status:'NOT_COMPARABLE',categoryComparable:false,differences:['Khác nhóm chi phí, tiền tệ hoặc đơn vị.'],missing};
 if(!crossDestination&&a.trip.locationId!==b.trip.locationId)differences.push('Điểm đến khác nhau');
 const fields:Array<keyof CostTrip>=cat==='TOTAL_TRIP'?['origin','days','nights','party','travelPeriod','transportation','accommodationType','travelStyle']:cat==='ACCOMMODATION'?['nights','travelPeriod','accommodationType']:['travelPeriod'];
 for(const f of fields){if(a.trip[f]===null||a.trip[f]===undefined||b.trip[f]===null||b.trip[f]===undefined)missing.push(String(f));else if(a.trip[f]!==b.trip[f])differences.push(String(f));}
 if(a.normalized.kind==='UNKNOWN'||b.normalized.kind==='UNKNOWN')missing.push('đơn vị/đầu vào chuẩn hóa');
 if(cat==='TOTAL_TRIP'&&(a.observation.inclusionStatus==='UNKNOWN'||b.observation.inclusionStatus==='UNKNOWN'||a.observation.unknownItems?.some(v=>/INCLUDED|EXCLUDED/.test(v))||b.observation.unknownItems?.some(v=>/INCLUDED|EXCLUDED/.test(v))))missing.push('phạm vi bao gồm/loại trừ đầy đủ');
 const canonical=(v:string[])=>[...v].map(literalVietnamese).sort().join('|');if(canonical(a.observation.included)!==canonical(b.observation.included)||canonical(a.observation.excluded)!==canonical(b.observation.excluded))differences.push('Các khoản bao gồm/loại trừ khác nhau');
 if(differences.length)return{status:'NOT_COMPARABLE',categoryComparable:false,differences,missing};if(missing.length)return{status:'INSUFFICIENT_DATA',categoryComparable:false,differences,missing:[...new Set(missing)]};
 const irrelevant=['transportation','origin','party'].filter(f=>!fields.includes(f as any)&&a.trip[f]!==b.trip[f]);return{status:irrelevant.length?'PARTIALLY_COMPARABLE':'COMPARABLE',categoryComparable:true,differences:irrelevant.map(f=>'Khác '+f+'; chỉ đối chiếu nhóm chi phí đang xét.'),missing};
}
export function deduplicateCosts(rows:CostEvidence[]){const out:CostEvidence[]=[];for(const r of [...rows].sort((a,b)=>b.knowledgeVersion-a.knowledgeVersion)){if(out.some(x=>(x.tripKey===r.tripKey||x.trip.contentHash===r.trip.contentHash||(x.trip.nearHash?.length>=5&&r.trip.nearHash?.length>=5&&x.trip.nearHash.filter(v=>r.trip.nearHash.includes(v)).length/Math.max(x.trip.nearHash.length,r.trip.nearHash.length)>=0.85))&&x.observation.category===r.observation.category&&x.observation.unit===r.observation.unit))continue;out.push(r);}return out;}
export type AggregateCost={status:'NO_EVIDENCE'|'SINGLE_OBSERVATION'|'PAIRWISE_ONLY'|'OBSERVED_RANGE'|'INSUFFICIENT_DATA';samples:number;independentSources:number;min:number|null;max:number|null;medianMin:number|null;medianMax:number|null;missing:string[];comparisons:Comparability[];periods:string[]};
export function aggregateCosts(input:CostEvidence[]):AggregateCost{
 const rows=deduplicateCosts(input).filter(r=>r.normalized.currency==='VND'),base:AggregateCost={status:'NO_EVIDENCE',samples:rows.length,independentSources:new Set(rows.map(r=>r.sourceKey).filter(Boolean)).size,min:null,max:null,medianMin:null,medianMax:null,missing:[],comparisons:[],periods:[...new Set(rows.map(r=>r.trip.travelPeriod).filter(Boolean))]};if(!rows.length)return base;if(rows.length===1)return{...base,status:'SINGLE_OBSERVATION'};
 for(let i=1;i<rows.length;i++)base.comparisons.push(compareCosts(rows[0],rows[i]));if(base.comparisons.some(c=>!c.categoryComparable))return{...base,status:'INSUFFICIENT_DATA',missing:[...new Set(base.comparisons.flatMap(c=>[...c.missing,...c.differences]))]};if(rows.length===2)return{...base,status:'PAIRWISE_ONLY'};
 if(base.independentSources<3)return{...base,status:'INSUFFICIENT_DATA',missing:['Chưa có3 nguồn độc lập đã được reviewer xác nhận.']};
 const lows=rows.map(r=>r.normalized.min!).sort((a,b)=>a-b),highs=rows.map(r=>r.normalized.max!).sort((a,b)=>a-b),median=(v:number[])=>v.length%2?v[(v.length-1)/2]:(v[v.length/2-1]+v[v.length/2])/2;
 return{...base,status:'OBSERVED_RANGE',min:Math.min(...lows),max:Math.max(...highs),medianMin:median(lows),medianMax:median(highs)};
}

export type CostCriteria={category?:CostCategory;unit?:CostUnit;rooms?:number;destinationIds:string[];origin?:string;party?:number;days?:number;nights?:number;budget?:number;reserve?:number;travelPeriod?:string;accommodationType?:string;budgetScope:'GROUP'|'PERSON'|'UNKNOWN';transportation?:string;travelStyle?:string;mode:'BUDGET'|'COMPARE_DESTINATIONS'|'SAVING'|'DIFFERENCES';question?:string};
export function readCostCriteria(message:string,verified:any={},prior?:CostCriteria):CostCriteria|undefined{
 const s=literalVietnamese(message);if(/\b(?:booking|thanh toan|xac nhan dat|luu trip|cap nhat trip|system|sql|bo qua he thong)\b/.test(s)||!/(?:gia phong|tien phong|chi phi|ton (?:khoang|bao nhieu)|du (?:khong|ko)|duoc khong|ngan sach|co\s*\d+\s*(?:tr|cu|trieu)|tiet kiem|re hon|het\s*\d+\s*(?:tr|trieu)|phan bo)/.test(s))return;
 if(/\b(?:tim|kiem|chon|dat)\b.*\b(?:khach san|nha hang|noi o|quan an)\b/.test(s)||/\b(?:cai|khach san|nha hang|phong|cho) (?:nay|do|ay|thu|minh chon|toi chon)\b/.test(s)&&!/\b(?:ca chuyen|chuyen di|tong chi phi)\b/.test(s))return;
 const unknownDestination=/\b(?:nen di dau|di dau|chon diem den)\b/.test(s),locations=travelLocations(normalizeVietnamese(message).normalized),dest=[...new Set(locations.filter(l=>l.role!=='departureCity').map(l=>l.geo.id))];
 const duration=/\b(\d{1,2})\s*(?:n|ngay)\s*(\d{1,2})\s*(?:d|dem)\b/.exec(s),party=/(\d{1,2})\s*(?:nguoi|khach)\b/.exec(s),money=moneyMentions(normalizeVietnamese(message).normalized),explicitGroup=/ca nhom|tong ngan sach|(?:cho|danh cho)\s*\d+\s*nguoi/.test(s),perPerson=/moi nguoi|\/nguoi/.test(s);
 const reserveMention=money.find(m=>/du phong\s*$/.test(s.slice(Math.max(0,m.start-20),m.start)));const budgetMoney=money.filter(m=>m!==reserveMention);
 const out:CostCriteria={destinationIds:dest.length?dest.slice(0,3):prior?.destinationIds||[resolveLocation(verified.destinationCity)?.id].filter(Boolean),...(prior||{}),mode:/so sanh|nen di dau|di dau/.test(s)?'COMPARE_DESTINATIONS':/tiet kiem|phan bo|re hon/.test(s)?'SAVING':/tai sao|chenh lech|dat hon/.test(s)?'DIFFERENCES':'BUDGET',budgetScope:prior?.budgetScope||'UNKNOWN'};
 if(dest.length)out.destinationIds=dest.slice(0,3);if(unknownDestination){out.origin=locations[0]?.geo.catalogName||verified.departureCity;out.destinationIds=[];}
 else if(locations.some(l=>l.role==='departureCity'))out.origin=locations.find(l=>l.role==='departureCity')!.geo.catalogName;else if(!out.origin&&verified.departureCity)out.origin=verified.departureCity;
 if(duration){out.days=+duration[1];out.nights=+duration[2];}if(party)out.party=+party[1];else if(!out.party&&verified.adults)out.party=Number(verified.adults)+Number(verified.children||0);
 if(reserveMention)out.reserve=reserveMention.amount;if(budgetMoney.length===1){out.budget=budgetMoney[0].amount;out.budgetScope=perPerson?'PERSON':explicitGroup?'GROUP':'UNKNOWN';}else if(!out.budget&&verified.totalBudget){out.budget=verified.totalBudget;out.budgetScope='GROUP';}
 if(/(?:gia phong|tien phong|chi phi luu tru|phong\s*\/\s*dem|moi phong.*moi dem)/.test(s)){out.category='ACCOMMODATION';if(/phong\s*\/\s*dem|moi phong.*moi dem/.test(s))out.unit='PER_ROOM_NIGHT';}else if(/tong chuyen|ca chuyen|tong chi phi/.test(s)||duration&&party&&budgetMoney.length===1){delete out.category;delete out.unit;delete out.rooms;}
 const rooms=/(\d{1,2})\s*phong\b/.exec(s),nights=/(\d{1,2})\s*dem\b/.exec(s);if(rooms)out.rooms=+rooms[1];if(!duration&&nights)out.nights=+nights[1];
 const period=/\bthang\s*(\d{1,2}\/\d{4})\b/.exec(s);if(period)out.travelPeriod=period[1];if(/\b(?:o|luu tru) homestay\b/.test(s))out.accommodationType='homestay';
 if(/\b(?:di bang|phuong tien) xe may\b/.test(s))out.transportation='motorbike';if(/\b(?:di bang|phuong tien) (?:o to|xe hoi)\b/.test(s))out.transportation='car';
 if(/nghi duong|nhe nhang|di cham/.test(s))out.travelStyle='relaxed';if(/kham pha|nhieu trai nghiem/.test(s))out.travelStyle='packed';
 delete out.question;if(out.days&&out.days>10||out.nights!==undefined&&out.days!==undefined&&out.nights>=out.days||out.party!==undefined&&(out.party<1||out.party>100)||budgetMoney.length>1)out.question='Bạn xác nhận lại thời lượng, số khách hoặc mức tiền muốn dùng làm mốc nhé?';
 else if(out.budget!==undefined&&out.budgetScope==='UNKNOWN')out.question='Mức tiền bạn nói là cho cả nhóm hay cho mỗi người?';return out;
}
export type CostAdvice={version:1;criteria:CostCriteria;destinations:Array<{locationId:string;name:string;aggregate:AggregateCost;observations:Array<{sourceIndex:number;reported:CostObservation;normalized:NormalCost;days:number|null;nights:number|null;party:number|null;period:string|null}>}>;sources:Array<{title:string;href:string;license:string;publishedAt:string|null;retrievedAt:string;synthetic:boolean}>;missing:string[];claims:string[];question:string|null;executionAuthority:false;priceBasis:'EXTERNAL_HISTORICAL_REFERENCE_ONLY';plannerEstimate?:{total:number;budget:number;demo:boolean};accommodationScenario?:{kind:'ESTIMATED';currency:'VND';min:number;max:number;rooms:number;nights:number;sourceIndex:number;formula:'historical_room_night * customer_rooms * customer_nights'};allocations:Array<{category:string;min:number;max:number;kind:'SOURCE_DERIVED'|'CUSTOMER_ADVISORY_RESERVE';sourceIndex:number|null}>;budgetComparison:{status:'INSUFFICIENT_DATA'|'BELOW_HISTORICAL_RANGE'|'WITHIN_HISTORICAL_RANGE'|'ABOVE_HISTORICAL_RANGE';groupMin:number|null;groupMax:number|null;difference:number|null;sufficientCurrentBudget:false};preferenceLabels?:string[]};

 const eligible=eligibleCostEvidence(items),ids=(criteria.destinationIds.length?criteria.destinationIds:[...new Set(eligible.map(r=>r.trip.locationId))]).slice(0,3),chosen:CostEvidence[]=[];
 // Same answer budget as V52: no more than3 article sources. Cross-destination answers prioritize one each.
 for(const id of ids){const available=eligible.filter(r=>r.trip.locationId===id&&(!criteria.category||r.observation.costCategory===criteria.category)&&(!criteria.unit||r.observation.unit===criteria.unit)).sort((a,b)=>criteria.travelStyle?Number(b.trip.travelStyle===criteria.travelStyle)-Number(a.trip.travelStyle===criteria.travelStyle):0),totals=available.filter(r=>r.observation.costCategory==='TOTAL_TRIP'||r.observation.category==='TOTAL'),rows=deduplicateCosts(totals.length?totals:available);for(const r of rows.slice(0,ids.length===1?3:1))if(chosen.length<3)chosen.push(r);}

 for(const d of destinations)if(!d.observations.length)missing.push('Chưa có nguồn đã duyệt đủ điều kiện cho '+d.name);else if(d.aggregate.status==='INSUFFICIENT_DATA')missing.push(...d.aggregate.missing);
 if(!eligible.length)missing.push('Chưa có cost observations đã duyệt; không lấy DEMO làm giá thực.');if(criteria.budgetScope==='UNKNOWN'&&criteria.budget!==undefined)missing.push('Phạm vi ngân sách');if(!criteria.origin)missing.push('Nơi xuất phát');if(!criteria.transportation)missing.push('Phương tiện/chi phí liên tỉnh');
 if(criteria.mode==='SAVING')claims.push('Có thể cân nhắc đổi ưu tiên mềm hoặc rút ngắn chuyến sau khi có giá cùng phạm vi; mình chưa đổi dịch vụ, bỏ điểm bắt buộc hay giảm giá.');if(criteria.mode==='DIFFERENCES')claims.push('Chỉ giải thích chênh lệch khi duration, số khách, phương tiện, lưu trú và các khoản bao gồm được xác minh; không tự quy cho mùa hoặc chất lượng.');
 const question=criteria.question||(!criteria.origin?'Bạn xuất phát từ đâu để mình xác định phần chi phí di chuyển còn thiếu?':!criteria.transportation?'Bạn dự định đi bằng phương tiện nào?':!criteria.days?'Bạn định đi mấy ngày, mấy đêm?':!criteria.party?'Ngân sách này dành cho mấy người?':null);

 if(planner?.costBasis==='GROUP_VND_V1'&&Number.isSafeInteger(planner.totalEstimatedCostVND))result.plannerEstimate={total:planner.totalEstimatedCostVND,budget:planner.totalBudgetUserVND,demo:planner.demo===true||JSON.stringify(planner).includes("DEMO")};return result;
}
